file(GLOB TPLG_DEPS
	${SOF_TOPOLOGY1_SOURCE_DIR}/platform/common/*.m4
	${SOF_TOPOLOGY1_SOURCE_DIR}/common/*.m4
	${SOF_TOPOLOGY1_SOURCE_DIR}/m4/*.m4
	${SOF_TOPOLOGY1_SOURCE_DIR}/sof/*.m4
	platform/*.m4
	*.m4
)

find_program(XARGS NAMES xargs PATHS ENV PATH NO_DEFAULT_PATH)

if(${XARGS} STREQUAL XARGS-NOTFOUND OR
    (DEFINED ENV{USE_XARGS} AND NOT $ENV{USE_XARGS} STREQUAL "yes")
    )
	set(USE_XARGS "no")
else()
	set(USE_XARGS "yes")
endif()

if(BUILD_TEST_TOPOLOGIES)
	set(_TESTS_ALL "ALL")
else()
	set(_TESTS_ALL "")
endif()

add_custom_target(tests ${_TESTS_ALL}
	COMMAND ${CMAKE_COMMAND} -E env
		SOF_TPLG_BUILD_OUTPUT=${CMAKE_CURRENT_BINARY_DIR}
		ALSATPLG_CMD=${ALSATPLG_CMD}
		LD_LIBRARY_PATH=${SOF_ALSA_LIB_DIR}:$ENV{LD_LIBRARY_PATH}
		USE_XARGS=${USE_XARGS}
		./tplg-build.sh
	DEPENDS ${TPLG_DEPS}
	WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
	VERBATIM
	USES_TERMINAL
)

add_custom_target(tests-clean
	COMMAND rm -f ${CMAKE_CURRENT_BINARY_DIR}/*.conf
	COMMAND rm -f ${CMAKE_CURRENT_BINARY_DIR}/*.tplg
	VERBATIM
	USES_TERMINAL
)
